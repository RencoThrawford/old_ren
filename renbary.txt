#This is my random sound function
#To use it, give it a array of sounds starting from 0, an entity to play the sounds from, and an index number
#input: Sounds - An array of sounds that the E2 will randomly choose from when you call this function
#input: Entity - The entity the sound will play on
#input: Index  - The index the sound will play at
function void randomSound(Sounds:array,Entity:entity,Index:normal)
{
    local SoundCount = Sounds:count()
    local RNG = randint(SoundCount)
    
    Entity:soundPlay(Index,0,Sounds[RNG,string])
}

#This is a function to setPos/Ang of a prop/entity to a hologram, then parent it
#input: Index  - The hologram index you would like to parent to
#input: Entity - The entity you would like to be parented
function void parentHolo(Index:normal,Entity:entity)
{
    local HoloE = holoEntity(Index)
    
    if(Entity:parent()!=HoloE&Entity:isValidPhysics())
    {
        local EPos = Entity:pos()
        local HPos = HoloE:pos()
        local PosDiff = EPos-HPos
        
        local EAng = Entity:angles()
        local HAng = HoloE:angles()
        local AngDiff = EAng-HAng
        
        if(inrange(PosDiff,vec(-0.01),vec(0.01))&inrange(AngDiff,ang(-0.01),ang(0.01))&Entity:parent()!=HoloE)
        {
            Entity:parentTo(HoloE)
            #print(Entity:type()+" parented!")
        }
        elseif(Entity:parent()!=HoloE)
        {
            Entity:setPos(HPos)
            Entity:setAng(HAng)
        }
    }
}


#This function checks if the gun is within acceptable bounds to fire; you should call this whenever you want to fire
#input: Gun            - The entity used to perform the interlock on
#input: AimPos         - Where we're aiming
#input: AngOffset      - Used if you're using an FCS
#input: Bound          - Angle in wich the gun has to be in to fire, if set to 0 it is forcibly disabled
#input: BlastRadiusInt - If set to 1, it will check infront of the gun, if the distance is less than or equal to the blast radius, it will not allow you to fire (useful for artillery)
#input: Override       - Manually overrides the interlock
#return: Returns wether or not the gun can fire with the set bound, or if it has been overriden
function number fireInterlock(Gun:entity,AimPos:vector,AngOffset:angle,Bound:normal,BlastRadiusInt:normal,Override:normal)
{
    #If we have chosen to disable it, or if we have the override codes, just return 1
    if(Bound == 0|Override) {return 1}
    
    #Or else we return something useful
    else
    {
        #/////HEADING INTERLOCK\\\\\#
        #Extracting the pitch and yaw offsets from AngOffset
        local PitchOffset = AngOffset[1]
        local YawOffset = AngOffset[2]
        
        #Getting the heading from the gun to where we're aiming
        local Heading = Gun:heading(AimPos)
        
        #Extracting the elevation and bearing, while adding our offsets
        local Elevation = Heading[1]+abs(PitchOffset)
        local Bearing = Heading[2]+YawOffset
        
        
        #/////BLAST RADIUS INTERLOCK\\\\\#
        local BlastRadius = Gun:acfBlastRadius()
        
        #Getting the hit position, should go out to 50 m
        local Forward = Gun:forward()
        local HitDist = (rangerOffset(1968.5,Gun:attachmentPos("muzzle")+Forward,Forward):distance())/39.37
            
        local BRI = (HitDist <= BlastRadius)*BlastRadiusInt
        
        
        #Finally, we return wether or not the gun is in acceptable bounds
        return (((abs(Bearing) <= Bound)&(abs(Elevation) <= Bound))&(!BRI))
    }
}

#This function manipulates the volume of a sound as the gun traverses, the closer to the AimPos+AngOffset it gets, the softer the sound gets; works kinda like Red's
#input: Gun        - The Gun entity for checking the heading on
#input: AimPos     - Where we're aiming
#input: AngOffset  - The offset outputted by an fcs, if you're not using one, just use ang(0)
#input: SoundIndex - The index of the sound to be manipulated (A number)
function void traverseSound(Gun:entity,AimPos:vector,AngOffset:angle,SoundIndex:normal)
{
    #Getting the heading of the gun to the AimPos, also subtracting the offset
    local Heading = Gun:heading(AimPos)-AngOffset
    
    #Adding the Elevation and Bearing, then dividing by 5, also absoluting because volume cannot be lower than 0
    local Length = abs(Heading[1]+Heading[2])/5
    
    
    #Finally we set the volume to the clamp of the length
    soundVolume(SoundIndex,clamp(Length,0,1))
}


#This function is kinda useless
#input: StartPos   - Where the lase will originate from
#input: AimPos     - Where the lase will be directed twords
#input: LastReturn - What this function last returned
#input: Curtime    - curtime()
#return: Returns an array with the position and range of the lase data
function array renLase(StartPos:vector,AimPos:vector,LastReturn:array,Curtime:normal)
{
    if(Curtime > (REN_LastLase + 0.5))
    {
        REN_LastLase = Curtime
        
        local Direction = (AimPos-StartPos):normalized()
        
        local Trace = rangerOffset(9999999999,StartPos+Direction,Direction)
        
        local Position = Trace:position()
        
        local Range = round(StartPos:distance(Position)/39.37)
        
        return array(Trace:position(),Range)
    }
    else
    {
        print("You cannot lase right now.")
        return LastReturn
    }
}


#This function just spawns a generic scope, no aimer or anything, you can make your own aimer with holos, or EGP
#input: HoloIndex - The hologram index at which the scope holograms will start to spawn at, they will be parented with the previous hologram
#return: Returns an array of the holograms used, the first entry will always be the scope tube, the second will always be the scope glass
function array ucGenericScope(HoloIndex:normal)
{
    #Getting information to spawn the holograms
    local Parent = HoloIndex-1
    local ParentE = holoEntity(Parent)
    
    #Preset our array of holograms
    local Indexes = array()
    
    
    #Scope Tube
    local A_Index = HoloIndex
    holoCreate(A_Index)
    holoModel(A_Index,"hq_tube_thin")
    holoAng(A_Index,ParentE:toWorld(ang(90,0,0)))
    holoPos(A_Index,ParentE:toWorld(vec(2.5,0,0)))
    holoParent(A_Index,Parent)
    holoColor(A_Index,vec(0))
    holoDisableShading(A_Index,1)
    holoShadow(A_Index,0)
    Indexes[1,normal] = A_Index
    
    #Scope Glass
    local A_Index = A_Index+1
    holoCreate(A_Index)
    holoModel(A_Index,"models/sprops/misc/domes/size_3/dome_36x18.mdl")
    holoAng(A_Index,ParentE:toWorld(ang(-90,0,0)))
    holoPos(A_Index,ParentE:toWorld(vec(8.5,0,0)))
    holoScale(A_Index,vec(0.33,0.33,0.01))
    holoParent(A_Index,Parent)
    holoMaterial(A_Index,"HUD/scope_sniper_ll")
    holoDisableShading(A_Index,1)
    holoShadow(A_Index,0)
    Indexes[2,normal] = A_Index
    
    
    #Return our array of holograms
    return Indexes
}


#This function updates a previously created generic scope with new textures and colours to somthing more suitable for FLIR
#input: ScopeHolos - Plug in the array that the generic scope generated
#input: FLIRActive - Either 1 or 0
function void ucGSFLIRUpdate(ScopeHolos:array,FLIRActive:normal)
{
    #Getting the ScopeTube and ScopeGlass from the ScopeHolos array, they should always be in this order
    local ScopeTube = ScopeHolos[1,normal]
    local ScopeGlass = ScopeHolos[2,normal]
    
    
    #Updating the hologram's colours, materials, and alphas
    #FLIR Active
    if(FLIRActive)
    {
        holoMaterial(ScopeGlass,"holograms/hologram")
        holoColor(ScopeGlass,vec(100))
        holoAlpha(ScopeGlass,150)
    }
    
    #FLIR Inactive
    else
    {
        holoMaterial(ScopeGlass,"HUD/scope_sniper_ll")
        holoColor(ScopeGlass,vec(255))
        holoAlpha(ScopeGlass,255)
    }
}

#This function updates the colour for the objective indicator
#input: AimerHolos - The array outputted by ucCreateDayAimer
#input: SCRLData   - The array outputted from SCRL when you recieve a transmission
function void ucUpdateObjectiveIndicatorCol(AimerHolos:array,SCRLData:array)
{
    #/////SCRL DATA\\\\\#
    #Get the order from the SCRLData array
    local Order = SCRLData[1,string]
    
    #Using Order, we search a precompiled table to get the colour
    local Color = UC_Col[Order,table]["color",vector]
    
    
    #/////HOLOGRAM MANIPULATION\\\\\#
    #Get the Objective Indicator's holo index
    local ObjInd = AimerHolos[9,normal]
    
    #Set the Objective Indicator's colour
    holoColor(ObjInd,Color)
}

function number ucRadarRange(StartHolo:normal,AimPos:vector,LastRange:normal)
{
    local StartHoloE = holoEntity(StartHolo)
    local Dir = (AimPos-StartHoloE:pos()):normalized()
    
    #local LastRangeUnits = LastRange*39.37
    
    local LastRangeUnits = 0
    
    local Size = UC_RadarRangeRatio * LastRangeUnits
    
    local TraceStartPos = StartHoloE:pos() + (Dir * 30)
    
    local Trace = rangerOffsetHull(UC_RadarRangeLength,TraceStartPos,Dir,vec(Size))
    
    if(Trace:entity())
    {
        return round(((Trace:position() - StartHoloE:pos()):length() + (Size / 2))/39.37)
    }
    else
    {
        return LastRange
    }
}
