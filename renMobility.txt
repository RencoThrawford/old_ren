Does mobility things

Type 0; Just gearbox and or steering control for a car or somthing
Type 1; Dual clutch tank steering, requires a dual-clutch gearbox
Type 2; Dual transfer case tank steering, two transfer cases being fed by a gearbox
Type 3; Dual gearbox tank steering, two gearboxes fed directly by an engine or a transfercase/differential
Type 4; Regnerative tank steering, regenerative steering case fed by a gearbox

!!!GEARBOX SHOULD BE THE GEARING GEARBOX!!!

    #[One of Red's functions to determine when to gear up or down
    Inputs:
        CurrentGear [NUMBER] - Current gear
        RPM [NUMBER] - RPM of the currently monitored engine
        PowerbandMin [NUMBER] - Minimum powerband of the currently monitored engine
        PowerbandMax [NUMBER] - Maximum powerband of the currently monitored engine
        PowerbandRedline [NUMBER] - Redline of the currently monitored engine
        Direction [NUMBER] - 1/-1 going forwads/backwards
        TransmissionGearRatios [ARRAY] - Array of gear ratios for main gearing gearbox
        CVT [NUMBER] - Are we using a CVT?
        MaximumThrottle [NUMBER] - Throttle currently used?
        ReversingGearbox [NUMBER] - Using a reversing gearbox?
        PitchAngle [NUMBER] - Pitch of the vehicle
    Return:
        [NUMBER] - 1/0/-1 shifting up, down, or not
    ]#
    function number evalShift(CurrentGear:number,RPM:number,PowerbandMin:number,PowerbandMax:number,PowerbandRedline:number,Direction:number,TransmissionGearRatios:array,CVT:number,MaximumThrottle:number,ReversingGearbox:number,PitchAngle:number){
        if(CVT) {
            if(!ReversingGearbox) {
                if((CurrentGear == 1) & (Direction == -1)) {
                    return 1
                } elseif((CurrentGear == 2) & (Direction == 1)) {
                    return -1
                } else {
                    return 0
                }
            }
            else {
                return 1 - CurrentGear
            }
        }

        local GearCount = TransmissionGearRatios:count()
        local CurrentRatio = TransmissionGearRatios[CurrentGear,number]

        #change of direction
        local CurrentRatioAsDirection = (ReversingGearbox) ? 1 : ((CurrentRatio < 0) ? -1 : 1)
        if(Direction != CurrentRatioAsDirection) {
            local LowestRatio = Direction
            local DesiredGear = 1
            for(N=1,GearCount) {
                local Ratio = TransmissionGearRatios[N,number]
                local SameDirection = ((Direction == 1) & (Ratio > 0)) | ((Direction == -1) & (Ratio < 0))
                if(SameDirection) {
                    DesiredGear = N
                }
            }

            return DesiredGear - CurrentGear
        }

        local DirectionalPitchAngle = PitchAngle * Direction
        local PitchRatio = clamp(DirectionalPitchAngle / 45,0,1)
        local IdealRPM = (DirectionalPitchAngle < 0) ? (PowerbandMax) : ((PowerbandMin * PitchRatio) + (PowerbandMax * (1 - PitchRatio)) )
        #determine if we can shift up
        local CanUpshift = (CurrentGear+1 <= GearCount) & (abs(TransmissionGearRatios[CurrentGear+1,number]) > abs(CurrentRatio)) & (DirectionalPitchAngle > -45)  #45d is pitch limit
        local CanDownshift = (CurrentGear-1 > 0) & (abs(TransmissionGearRatios[CurrentGear-1,number]) < abs(CurrentRatio))
        #evaluation of next shift point
        local UpRatio = TransmissionGearRatios[CurrentGear + 1,number]
        local DownRatio = TransmissionGearRatios[CurrentGear - 1,number]
        local CurrentRatio = TransmissionGearRatios[CurrentGear,number]
        local OutRPM = (RPM * CurrentRatio)
        local UpRPM = OutRPM / UpRatio
        local DownRPM = OutRPM / DownRatio
        local UpDifference = abs(UpRPM - IdealRPM)  #the difference between ideal RPM, and the RPM engine would be at in higher gear
        local DownDifference = abs(DownRPM - IdealRPM)  #the difference between ideal RPM, and the RPM engine would be at in lower gear
        local CurrentDifference = abs(RPM - IdealRPM) #the difference between ideal RPM and the current RPM

        #heuristic estimate of roughly how high the rev can go
        #TODO: replace with a more scientific system
        local ThrottleRPM = (MaximumThrottle == 100) ? (PowerbandRedline) : (PowerbandMax + ((PowerbandRedline - PowerbandMax) * (MaximumThrottle / 200)))

        local MaxAllowedRPM = clamp(ThrottleRPM,PowerbandMax,(PowerbandMax * 0.5) + (PowerbandRedline * 0.5))

        if((UpDifference < CurrentDifference) | (RPM >= MaxAllowedRPM)) {
            if(CanUpshift) {
                return 1
            }
        } elseif((DownDifference < CurrentDifference) & (DownRPM < PowerbandRedline) & (DownRPM < MaxAllowedRPM)) {
            if(CanDownshift) {
                return -1
            }
        }

        return 0
    }

    #[One of Red's functions to help control steering of tracked vehicles using a PID
    Inputs:
        Input [NUMBER] - Error
        Settings [ARRAY] - arrray(P,I,D,Clamp)
        State [ARRAY] - The array to be used to store stuff for the PID
        Deadzone [NUMBER] - Deadzone for proportional
    Return:
        [NUMBER] - Modifier to be added to steering input
    ]#
    function number steerPID(Input:number,Settings:array,State:array,Deadzone:number) {
        local Hold = Deadzone < 0
        local Proportional = (Hold) ? (Input * (abs(Input) > Deadzone)) : (Input)
        local ForceError = (Input - State[1,number]) * 0.1  #force feedback
        local Integral = clamp(State[1,number] + (Input + ForceError),-Settings[4,number],Settings[4,number])
        local Derivative = Input - State[2,number]

        local ForceSum = (Proportional*Settings[1,number]) + (Integral*Settings[2,number]) + (Derivative*Settings[3,number])
        local PID = (!Hold) ? ((abs(ForceSum) > (Deadzone / 2)) ? (ForceSum) : (0)) : (ForceSum)

        State[1,number] = Integral
        State[2,number] = Input
        State[3,number] = PID

        return PID
    }

    #[One of Red's functions to generate an array of all gear ratios of a gearbox
    Inputs:
        Transmission [ENTITY] - The gearbox to generate the array for
    Return:
        [ARRAY] - Array of gear ratios

    function array getGearRatios(Transmission:entity) {
        local Array = array()
        for(N=1,Transmission:acfNumGears()) {
            local Ratio = Transmission:acfGearRatio(N)
            Array[N,number] = int(Ratio * 100) / 100    #fixes some shit with wiremod
        }
        return Array
    }

    #[One of Red's functions to check if an entity is makesphericaled
    Inputs:
        [ENTITY] - The entity to be checked if spherical
    Return:
        [NUMBER] - Is the entity spehcial?
    ]#
    function number entity:checkSphericalWheel()
    {
        local Radius = This:radius()
        if(Radius < 10) {return 0}
        local Box = This:boxSize()
        if(Box[1] == Box[2])
        {
            if(Box[2] == Box[3])
            {
                if(Box[3] == Box[1])
                {
                    return 1
                }
                else
                {
                    return 0
                }
            }
            else
            {
                return 0
            }
        }
        else
        {
            return 0
        }
    }

    #[A simple pid for controlling an engine's throttle to maintain a certain speed
    Inputs:
        [ARRAY] - The array last returned
        CurSpeed [NUMBER] - Our current speed
        MaxSpeed [NUMBER] - Maximum allowable speed
        Frequency [NUMBER] - How many miliseconds between executions
        Settings [VECTOR] - Settings for a pid vec(Kp,Ki,Kd)
    Return:
        [ARRAY] - And array of array(P+I+D,LastError,I)
    ]#
    function array array:pidSpeedGovernor(CurSpeed:number,MaxSpeed:number,Frequency:number,Settings:vector)
    {
        local Timestep = Frequency/1000
        local Error = MaxSpeed - CurSpeed
        local I = clamp(This[3,number] + Error,-50,50)*Timestep
        local D = (Error - This[2,number])/Timestep
        return array((Settings[1]*Error)+(Settings[2]*I)+(Settings[3]*D),Error,I)
    }

                    {
                        timer("steering",STEERING_FREQUENCY)
                        timer("gearing",GEARING_FREQUENCY)
                        timer("misc",MISC_FREQUENCY)
                        timer("check",CHECK_FREQUENCY)

                        Gearbox:soundPlay("whine",0,gearboxSound(Gearbox))
                        soundVolume("whine",0)

                        Engines[1,entity]:soundPlay("eidle",0,IDLESOUND)
                    }

                    else
                    {
                        stoptimer("misc")
                        stoptimer("gearing")
                        stoptimer("steering")
                        stoptimer("check")

                        soundStop("whine")

                        soundStop("eidle")
                        soundVolume("tbss",0)
                    }
                }
            break

            case RenKL_KEYS["HORN",string],
                if(Saftey&Pressed) {soundPlay("horn",0,HORNSOUND)} else{soundStop("horn")}
            break

            case RenKL_KEYS["TOGGLE_MOUSE",string],
                if(Started&Pressed){Mouse_Steer = !Mouse_Steer soundSwitch() operatorNotify("Mouse steering "+ACTIVATIONS[Mouse_Steer,string]+".")}
            break

            case RenKL_KEYS["TOGGLE_THROTTLE_LOCK",string],
                if(Pressed) {ThrottleLock = !ThrottleLock Throttle = ThrottleLock*100 soundSwitch() operatorNotify("Throttle lock "+ACTIVATIONS[ThrottleLock,string]+".")}
            break
            
            case RenKL_KEYS["TOGGLE_FUEL",string],
                if(Pressed)
                {
                    if(Saftey) {operatorNotify((Engines[1,entity]:acfFuelLevel()*100)+"% fuel remaining. (Fuel "+ACTIVATIONS[Fuel_On,string]+")")}
                    else{Fuel_On = !Fuel_On FUELS:acfActive(Fuel_On) soundSwitch() operatorNotify("Fuel use "+ACTIVATIONS[Fuel_On,string]+".")}
                }
            break
            
            case RenKL_KEYS["VIEW_SWITCH",string],
                if(Pressed)
                {
                    if(DriverSeat:driver()==Operator) {DriverCamActive = !DriverCamActive DriverCam["Activated",number] = DriverCamActive soundSwitch()}
                }
            break

            default,
                WS = Operator:keyPressed(RenKL_KEYS["FORWARD",string])-Operator:keyPressed(RenKL_KEYS["BACKWARD",string]) #Backwards or forwards?
                if(changed(WS)&WS) {LastWS = WS} #The last direction we were going in
                if(LastWS==1) {Transmission_Mode = "FORWARD"} elseif(LastWS==-1) {Transmission_Mode = "REVERSE"} else{TRANSMISSION_MODE = "NEUTRAL"} #I guess this is how I do this?

                if(!Mouse_Steer) {AD = Operator:keyPressed(RenKL_KEYS["LEFT",string])-Operator:keyPressed(RenKL_KEYS["RIGHT",string])} #If we're not using mouse steer use the usual left/right keys
                
                if((Saftey&(Key == RenKL_KEYS["BREAKS",string]))&Pressed) {ParkingBrake = !ParkingBrake soundSwitch() operatorNotify("Parking Brake "+ACTIVATIONS[ParkingBrake,string]+".")}
                Brakes_Applied = Operator:keyPressed(RenKL_KEYS["BREAKS",string])|ParkingBrake #We're using the Brakes if we're using the Brakes, or the parking Brake
            break
        }
    }
}

elseif(clk("steering"))
{
    timer("steering",STEERING_FREQUENCY)
    local Aim = Aimpos
    if(DriverSeat:driver()) {Aim = DriverCam["HitPos",vector]}
    if(Mouse_Steer) {AD = clamp(((-BASE:bearing(Aim)*LastWS)/(MAXSTEERANGLE*(SLAVEUSE+1))),-1,1)}

    Steering_Input += clamp(AD-Steering_Input,-INPUTSMOOTHER,INPUTSMOOTHER)

    BaseAng = BASE:angles()

    Clutch = Brakes_Applied|(RPM < Powerband[2])
    Brake = Brakes_Applied*BRAKEFORCE

    if(SLAVEUSE)
    {
        local DesiredAng = MAXSTEERANGLE*Steering_Input

        local SlaveAng = ang(0,clamp(DesiredAng,-MAXSTEERANGLE,MAXSTEERANGLE),0)

        if(F1) {F1:setAng(BaseAng+SlaveAng)}
        if(F2) {F2:setAng(BaseAng+(SlaveAng/MIDDLESTEERDIVISOR))}
        if(R1) {R1:setAng(BaseAng-(SlaveAng/MIDDLESTEERDIVISOR))}
        if(R2) {R2:setAng(BaseAng-SlaveAng)}

        if(!F1:isFrozen())
        {
            F1:propFreeze(1)
            F2:propFreeze(1)
            R1:propFreeze(1)
            R2:propFreeze(1)
        }
    }

    if(TYPE)
    {
        BaseAngVel = BASE:angVel()

        local TrackError = ((Steering_Input*MAXYAWVEL)-(BaseAngVel:yaw()*LastWS))

        local SteeringPID = steerPID(TrackError,PIDSETTING,PidState,TRACKDEADZONE)

        TrackSteering = clamp(Steering_Input+SteeringPID,-1,1)

        CrossLatch = (abs(TrackSteering) < 0.05) & (abs($TrackSteering) < 0.1)

        if(CrossLatch)
        {
            LeftClutch=RightClutch = Clutch
            LeftBrake=RightBrake = Brake
            LeftLatch=RightLatch = Brakes_Applied
            SteerRate=0

            if(TYPE != 3) {LeftGear=RightGear = 1}
            else{LeftGear=RightGear = Gear}
        }

        else
        {
            if(Brakes_Applied)
            {
                if(inrange(TYPE,2,4)&abs(TrackSteering))
                {
                    if(inrange(TYPE,2,3)) {LeftClutch=RightClutch=LeftBrake=RightBrake=LeftLatch=RightLatch = inrange(TrackSteering,-0.125,0.125)}
                    else{LeftClutch=Rightclutch=1 LeftBreak = 0 RightBreak = 0 LeftLatch=RightLatch = 0}

                    if(TYPE == 2)
                    {
                        if(inrange(TrackSteering,-0.125,0.125)) {LeftGear=RightGear = 1}
                        else{LeftGear = 1+(TrackSteering>0) RightGear = 1+(TrackSteering<0)}
                    }

                    elseif(TYPE == 3)
                    {
                        if(inrange(TrackSteering,-0.125,0.125)) {LeftGear=RightGear = 1}
                        else{LeftGear = 1+(TrackSteering>0) RightGear = 1+(TrackSteering<0)}
                    }

                    elseif(TYPE == 4)
                    {
                        SteerRate = -TrackSteering
                        LeftClutch=Rightclutch = 1
                        LeftBrake=RightBrake = 0
                        LeftLatch=RightLatch = 0
                    }
                }
            }

            else
            {
                LeftClutch = abs(clamp(TrackSteering,0,1))
                RightClutch = abs(clamp(TrackSteering,-1,0))

                LeftBrake = LeftClutch*BRAKEFORCE
                RightBrake = RightClutch*BRAKEFORCE

                LeftLatch = inrange(TrackSteering,0.9,1)
                RightLatch = inrange(TrackSteering,-1,-0.9)

                if(TYPE != 3) {LeftGear=RightGear = 1}
                else{LeftGear=RightGear = Gear}
            }
        }
    }
}

elseif(clk("gearing"))
{
    timer("gearing",GEARING_FREQUENCY)

    RPM = Engines[1,entity]:acfRPM()

    Gear += evalShift(Gear,RPM,Powerband[2],Powerband[3],Powerband[4],ThrottleLock ? LastWS : WS,GEARRATIOS,ISCVT,Throttle,0,BASE:angles():pitch())
}

elseif(clk("misc"))
{
    timer("misc",MISC_FREQUENCY)

    soundVolume("eidle",clamp((Powerband[1]/RPM),0,1),1)

    if(Gearbox:isValid())
    {
        if(changed(Gear)) {Gearbox:soundPlay("mgear",0.3,"acf_extra/vehiclefx/trans/default_shift.wav") soundVolume("mgear",0.5)}
        TransPitch:gearboxSoundUpdate("whine",RPM,Powerband[4],Clutch)
    }

    if(inrange(TYPE,2,3))
    {
        if(LeftGearbox:isValid()&changed(LeftGear)) {Gearbox:soundPlay("lgear",0.3,"acf_extra/vehiclefx/trans/default_shift.wav") soundVolume("lgear",0.5)}
        if(RightGearbox:isValid()&changed(RightGear)) {Gearbox:soundPlay("rgear",0.3,"acf_extra/vehiclefx/trans/default_shift.wav") soundVolume("rgear",0.5)}
    }

    tbss_update((TYPE==0)|F1:isValid(),curtime(),abs(Velocity:length()))

    Speed_Ratio = clamp(toUnit("km/h",Velocity:length())/MAXSPEED,0,1)

    if(GOVERNORMODE) {Governor = Governor:pidSpeedGovernor(toUnit("km/h",Velocity:length()),MAXSPEED,MISC_FREQUENCY,GOVERNORSETTING)}
    local Adjust = (Governor[1,number] < 0) ? clamp(abs(Governor[1,number]),0,1) : 0
    if(TYPE) {Throttle = ((WS|AD|ThrottleLock)-Adjust)*100} else{Throttle = ((WS|ThrottleLock)-Adjust)*100}

    msgBusOut()
}

elseif(clk("check"))
{
    timer("check",CHECK_FREQUENCY)
    local NewEngineCount = 0
    for(C=1,Engines:count()) {if(Engines[C,entity]:isValid()) {NewEngineCount++} else{Engines:removeEntity(C)}}
    if(NewEngineCount < EngineCount)
    {
        soundExplode() operatorNotify("Engine destroyed!")
        local NewEngine = Engines[1,entity]
        if(Engines[1,entity]:isValid()) {Engines[1,entity]:updateInfo()} else{operatorNotify("Mobility disabled!")}
    }
    EngineCount = NewEngineCount
}

elseif(clk("operator_query"))
{
    timer("operator_query",(Operator) ? (OPERATOR_QUERY_FREQUENCY) : (OPERATOR_QUERY_FREQUENCY >> 1))
    if(DriverSeat:driver()) {Operator = DriverSeat:driver()}
    elseif(Gunner) {Operator = Gunner}
    else{Operator = noentity()}
    if(changed(Operator)) {DriverCam["Activated",number] = DriverCamActive&DriverSeat:driver():isPlayer()}
}
