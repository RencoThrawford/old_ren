#[This function sets the visibility of an array of hologram indexes to the player entity and other players accordingly
Inputs:
    [ENTITY] - The player you wish to have holograms visible for
    Indexes [ARRAY] - The array of hologram indexes you wish to have their visibility changed
    Invert [NUMBER] - Inverts who can/cannot see the holograms
]#
function void entity:renOLVis(Indexes:array,Invert:number)
{
    local OtherPlayers = array()

    foreach(K,P:entity = players()) {if(P!=This){OtherPlayers:pushEntity(P)}}

    for(I=1,Indexes:count())
    {
        local CurIndex = Indexes[I,number]
        holoVisible(CurIndex,OtherPlayers,Invert)
        holoVisible(CurIndex,This,!Invert)
    }
}

#[This function updates 3 hologram numbers
Inputs
    [NUMBER] - The starting index for the numbers
    Number [NUMBER] - The number to be displayed
]#
function void number:renOLUpdateDigits(Number:number)
{
    local RoundedNumber = min(int(Number),999)
    local StringNumber = RoundedNumber:toString()

    if(changed(RoundedNumber))
    {
        if(RoundedNumber > 100)
        {
            holoModel(This,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(1)+".mdl")
            holoModel(This+1,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(2)+".mdl")
            holoModel(This+2,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(3)+".mdl")
        }
        elseif(RoundedNumber > 10)
        {
            holoModel(This,"models/sprops/misc/alphanum/alphanum_0.mdl")
            holoModel(This+1,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(1)+".mdl")
            holoModel(This+2,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(2)+".mdl")
        }
        else
        {
            holoModel(This,"models/sprops/misc/alphanum/alphanum_0.mdl")
            holoModel(This+1,"models/sprops/misc/alphanum/alphanum_0.mdl")
            holoModel(This+2,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(1)+".mdl")
        }
    }
}

#[This function sets the camo color of the sight+E2 to whatever you specify, can also set material if the camo color only does color
Just call this after you've built the sight
Inputs:
    Colour [STRING] - The name of the color things will be coloured to
    Material [STRING] - The material that will be used if the color specified only does color
]#
function void renOLCamoSight(Colour:string,Material:string)
{
    if(RenOL_SIGHTCAMO:count())
    {
        local Lower = Colour:lower()
        local Color = RenOL_CAMOCOLS[Colour,vector]
        local Mat = Material
        if(!RenOL_CAMONOT[Lower,number]) {Mat = RenOL_CAMOMAT}

        for(H=1,RenOL_SIGHTCAMO:count())
        {
            holoColor(RenOL_SIGHTCAMO[H,number],Color)
            holoMaterial(RenOL_SIGHTCAMO[H,number],Mat)
        }

        entity():setColor(Color)
        entity():setMaterial(Mat)
    }
}

#[This function updates the sight (Will only do anything if we're using a rotating sight)
Inputs:
    Aimpos [VECTOR] - Where we're aiming so the sight will point at that
]#
function void renOLUpdateSight(Aimpos:vector)
{
    switch(RenOL_SIGHTMODE)
    {
        #Two Axis (Azimuth & Elevation) {Left/Right, Up/Down}
        case 1,
            local AziEnt = holoEntity(RenOL_ROTSIGHTAZI)
            local EleEnt = holoEntity(RenOL_ROTSIGHTELE)

            #Probably wont work
            holoAng(RenOL_ROTSIGHTAZI,AziEnt:toWorld(ang(0,AziEnt:bearing(Aimpos)*-0.5,0)))

        break

        #One Axis (Azimuth) {Left/Right}
        case 2,

            holoAng(RenOL_ROTSIGHTAZI,AziEnt:toWorld(ang(0,AziEnt:bearing(Aimpos)*-0.5,0)))
        break

        #One Axis (Elevation) {Up/Down}

            holoAng(RenOL_ROTSIGHTELE,EleEnt:toWorld(ang(EleEnt:elevation(Aimpos)*-0.5,0,0)))

    }

{
    if(RenOL_SCOPEPARENTE:isValid())
    {
        #Range update
        if((RenOL_AimerTimer % 16) == 1)
        {
            (RenOL_AIMERI+9):renOLUpdateDigits(Range)
        }

        #Objective indicator update
        if((RenOL_AimerTimer % 8) == 1)
        {
            if(SCRLData:count())
            {
                
            }
        }

        #Gun readiness indicator update
        if((RenOL_AimerTimer % 4) == 1)
        {
            holoColor(RenOL_AIMERI+13,GunReady ? RenOL_GREEN : RenOL_RED)
        }

        if(RenOL_AimerTimer) {RenOL_AimerTimer = RenOL_AimerTimer-1} else{RenOL_AimerTimer = 20}

}

#[This function updates the visbility for the sight, scope, and aimer
Inputs:
    [ENTITY] - The person who will see the scope and aimer and not see the sight in first person
    CamOn [NUMBER] - Is the camera on
    ThirdPerson [NUMBER] - Is the gunner in third person?
]#
function void entity:renOLUpdateVisibility(CamOn:number,ThirdPerson:number)
{
    local Person = noentity()

    if(CamOn&(!ThirdPerson)) {Person = This}

    Person:renOLVis(RenOL_SIGHTREFS,1)
    Person:renOLVis(RenOL_SCOPEREFS,0)
    Person:renOLVis(RenOL_AIMERREFS,0)
}
