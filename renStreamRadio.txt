    #[Updates the numerical display if present, assumes you are using 3 digits
    Inputs:
        [NUMBER] - The hologram index of the leftmost number
        Number [NUMBER] - The number that will be displayed
    ]#
    function void number:updateNumeralDisplay(Number:number)
    {
        local RoundedNumber = min(int(Number),999)
        local StringNumber = RoundedNumber:toString()

        if(changed(Number))
        {
            if(RoundedNumber >= 100)
            {
                holoModel(This,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(1)+".mdl")
                holoModel(This+1,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(2)+".mdl")
                holoModel(This+2,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(3)+".mdl")
            }

            elseif(RoundedNumber > 10)
            {
                holoModel(This,"models/sprops/misc/alphanum/alphanum_0.mdl")
                holoModel(This+1,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(1)+".mdl")
                holoModel(This+2,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(2)+".mdl")
            }

            else
            {
                holoModel(This,"models/sprops/misc/alphanum/alphanum_0.mdl")
                holoModel(This+1,"models/sprops/misc/alphanum/alphanum_0.mdl")
                holoModel(This+2,"models/sprops/misc/alphanum/alphanum_"+StringNumber:index(1)+".mdl")
            }
        }
    }

    #[Updates the alphanumeric display if present
    Inputs:
        [VECTOR2] - The index of the leftmost letter/number, and the length of the display [INDEX,LENGTH]
        String [ARRAY] - The string that will be displayed
    ]#
    function void vector2:updateAlphanumDisplay(String:array)
    {
        local Index = This[1]
        local Length = This[2]

        for(L=1,Length)
        {
            local Character = String[L,string]:lower()

            switch(Character)
            {
                case "",
                case " ",
                    holoAlpha(Index+L,0)
                break

                default,
                    holoColor(Index+L,DISPLAYCOLOUR)
                    holoModel(Index+L,"models/sprops/misc/alphanum/alphanum_"+Character+".mdl")
                break
            }
        }
    }

    #[Formats seconds into a vector (SS,MM,HH)
    #Inputs:
        [NUMBER] - Time in seconds
    #Return:
        [VECTOR] - Time in seconds, minutes, hours vec(S,M,H)
    ]#
    function vector number:formatTime()
    {
        local Seconds = floor(This%60)
        local Minutes = floor(This/60)
        local Hours = floor(Minutes/60)
        Minutes = floor(Minutes%60)
        
        return vec(Seconds,Minutes,Hours)
    }


    #//STREAM\\#
    #[Creates a stream on the E2 with the specified index, URL, and volume
    #Inputs:
        [NUMBER] - Index of the stream
        URL [STRING] - The URL of the stream
        Volume [NUMBER] - The volume (0-1)
    ]#
    function void number:streamCreate(URL:string,Volume:number)
    {
        local Ran = 0
        #ifdef entity:streamStart(number,string,number,number)
            entity():streamStart(This,URL,Volume,100)
            Ran = 1
        #endif
        #ifdef entity:streamStart(number.number,string)
            if(!Ran)
            {
                entity():streamStart(This,Volume,URL)
                Ran = 1
            }
        #endif
    }

    #[Destroys a stream of the specified index
    #Inputs:
        [NUMBER] - Index of the stream
    ]#
    function void number:streamDestroy()
    {
        #ifdef streamStop(number)
            streamStop(This)
        #endif
    }

    #[Changes the volume of the stream
    #Inputs:
        [NUMBER] - Index of the stream
        Volume [NUMBER] - The volume (0-1)
    ]#
    function void number:streamVolume(Volume:number)
    {
        #ifdef streamVolume(number,number)
            streamVolume(This,Volume)
        #endif
    }

    #[Returns how fast we can create streams, used for anti-spam
    #Return:
        [NUMBER] - Seconds + 0.25
    ]#
    function number streamMinInterval()
    {
        local Return = 0

        #ifdef streamLimit()
            Return = streamLimit()+0.25
        #endif

        return Return
    }


    #//MISC\\#
    #[Notifies the owner of a message
    Inputs:
        Message [STRING] - The message that will be hinted to the owner
    ]#
    function void ownerNotify(Message:string)
    {
        hint("[renStreamRadio] "+Message,5.5)
    }

    #[Tells the owner a message
    Inputs:
        Message [STRING] - The message that will be printed to the owner's chat
    ]#
    function void ownerTell(Message:string)
    {
        print("[renStreamRadio] "+Message)
    }

    #[Prints a message into the owner's console
    Inputs:
        Message [STRING] - The message that will be printed to the owner's console
    ]#
    function void ownerConsole(Message:string)
    {
        print(_HUD_PRINTCONSOLE,"[renStreamRadio] "+Message)
    }



    #/////FINALIZATION\\\\\#
    #//MISC\\#
    runOnChat(1)

    #Force a track update for the display
    timer("track_update",25)


    #//VERSION\\#
    VERSION = "V1.0.6"
    setName("renStreamRadio "+VERSION)
}



elseif(chatClk(owner()))
{
    local Said = owner():lastSaid():lower()
    local Split = Said:explode(" ")
    local Token = Split:string(1)
    local Command = Split:string(2)
    local Arg1 = Split:string(3)
    local Arg1Num = Arg1:toNumber()

    if(Token == "radio")
    {
        switch(Command)
        {
            #Manual; provides a description of the provided command.
            case "man",
            case "manual",
                Arg1 = Arg1:left(3)
                if(MANUAL:exists(Arg1)) {ownerTell(Arg1+": "+MANUAL[Arg1:toNumber(),string])}
                else{ownerTell("Command "+Arg1+" not found, please use hel for a list of available commands.")}
            break

            #Help; provides a list of usable commands.
            case "hel",
            case "help",
                ownerTell("Commands: man, hel, ver, res, pre, sto, vol, abo, rep, tra, aut, ran, ski.")
            break

            #Version; prints out the version, credits, and a hash
            case "ver",
            case "version",
                ownerTell(VERSION)
                ownerConsole(VERSION+" By: Renco; "+hashNoComments()+".")
            break

            #Reset; resets the E2
            case "res",
            case "reset",
                ownerNotify("Resetting.")
                STREAMINDEX:streamDestroy()
                reset()
            break

            #Preset; plays the specified track
            case "pre",
            case "pla",
            case "preset",
            case "play",
            case "track",
                if(Autoplay) {ownerNotify("Currently in autoplay mode, please use sto to stop autoplay.")}

                else
                {
                    if(TRACKS:exists(Arg1Num))
                    {
                        CurTrackIndex = Arg1Num
                        timer("track_update",15)
                    }
                    else
                    {
                        ownerNotify("Cannot find track number: "+Arg1Num+".")
                    }
                }
            break

            #Stop; stops the radio, and autoplay
            case "sto",
            case "stop",
                ownerNotify("Stopping radio.")

                stoptimer("track_update") stoptimer("autoplay_update")

                timer("track_update",15)

                CurTrackIndex=Autoplay=Random = 0

                AutoplayPlayed = array()
            break

            #Volume; sets the volume for new and current streams
            case "vol",
            case "volume",
                if(Arg1Num) {CurVolume = Arg1Num/100 STREAMINDEX:streamVolume(CurVolume) ownerNotify("Setting stream volume to "+Arg1Num+"%.")}
                else{ownerNotify("Current stream volume "+CurVolume*100+"%.")}
            break

            #About; prints information about the provided or current track
            case "abo",
            case "about",
                if(Arg1Num&TRACKS:exists(Arg1Num)) {ownerTell(TRACKS[Arg1Num,table]:toString())}
                else{ownerTell(CurTrackTable:toString())}
            break

            #Replay; replays the last played track
            case "rep",
            case "replay",
                if(Autoplay) {ownerNotify("Currently in autoplay mode, please use sto to stop autoplay.")}
                else{ownerNotify("Replaying last track.") timer("track_update",25)}
            break

            #Tracks; prints infomration about every track available
            case "tra",
            case "tracks",
                local Total = 0

                for(T=1,TRACKCOUNT) {Total = Total+TRACKS[T,table]["track_dura",number]}

                local Time = Total:formatTime()

                ownerTell(TRACKCOUNT+"; "+AUTOPLAYTRACKCOUNT+" autoplayable, H"+Time[3]+" M:"+Time[2]+" S:"+Time[1])
            break

            #Autoplay; plays tracks in order
            case "aut",
            case "auto",
            case "autoplay",
                if(Autoplay) {ownerNotify("Already in autoplay mode, please use sto to stop autoplay.")}
                else{ownerNotify("Beginning autoplay.") timer("autoplay_update",100) Autoplay = 1 Random = 0}
            break

            #Random; plays a random track, remembering last tracks played
            case "ran",
            case "shu",
            case "random",
            case "shuffle",
                if(Autoplay) {ownerNotify("Already in autoplay mode, please use sto to stop autoplay.")}
                else{ownerNotify("Beginning random autoplay.") timer("autoplay_update",100) Autoplay=Random = 1}
            break

            #Skip; forces autoplay to choose a new track
            case "ski",
            case "skip",
                if(Autoplay) {ownerNotify("Skipping current track.") stoptimer("autoplay_update") timer("autoplay_update",100)}
                else{ownerNotify("Currently not in autoplay mode, please use aut or ran to begin autoplay.")}
            break

            #They got it wrong, tell them
            default,
                ownerNotify("Command "+Command+" not found, please use hel for a list of available commands.")
            break
        }

        hideChat(1)
    }
}



#PLAYING TRACKS
elseif(clk("track_update"))
{
    #Check if we're gonna trigger the antispam
    if((curtime()-LastPlayTime)>streamMinInterval())
    {
        #Used for antispam
        LastPlayTime = curtime()

        #Stop the last stream, mostly for replay
        STREAMINDEX:streamDestroy()

        #Update the current track table
        CurTrackTable = TRACKS[CurTrackIndex,table]
        local Time = CurTrackTable["track_dura",number]:formatTime()

        #Play the new track if its non zero and exists
        if((CurTrackIndex>0)&TRACKS:exists(CurTrackIndex))
        {
            STREAMINDEX:streamCreate(CurTrackTable["track_url",string],CurVolume)

            ownerNotify("Playing track: "+CurTrackIndex+"; "+CurTrackTable["track_name",string]+".")

            #Start autoplay again if we're using that, also avoid antispam
            if(Autoplay)
            {
                timer("autoplay_update",(max(CurTrackTable["track_dura",number],streamMinInterval())+TRACKPADDING)*1000)
            }
        }

        if(NUM) {NUM:updateNumeralDisplay(CurTrackIndex)}
        if(ALPHANUM[2]) {timer("display_update",5) ScrollIndex = 1 Scroll = array()}

        #Set the name
        setName(
            "renStreamRadio "+VERSION+"\n"+
            "Currently Playing: "+CurTrackTable["track_arti",string]+" - "+CurTrackTable["track_name",string]+"\n"+
            "Runtime; H:"+Time[3]+" M:"+Time[2]+" S:"+Time[1]
        )
    }

    #Try again
    else
    {
        stoptimer("track_update")

        local Wait = streamMinInterval()

        timer("track_update",Wait*1000)

        ownerNotify("Anti-spam, waiting "+Wait+" seconds.")
    }
}



#AUTOPLAY
elseif(clk("autoplay_update"))
{
    #Random autoplay mode
    if(Random)
    {
        #Stop autoplay when we've played everything
        if(AutoplayPlayed:count()+1 > AUTOPLAYTRACKCOUNT) {Autoplay=Random = 0 AutoplayPlayed = array() stoptimer("autoplay_update") ownerNotify("Reached end of track list, stopping.")}

        else
        {
            #Choose a random number
            local RandomInt = randint(AUTOPLAYTRACKCOUNT)

            #We've played this track before, try again
            if(AutoplayPlayed:exists(RandomInt)) {stoptimer("autoplay_update") timer("autoplay_update",25) ownerConsole("Already played!")}

            #We havent played this track, add it to the list and start the musak
            else{stoptimer("autoplay_update") AutoplayPlayed:pushNumber(RandomInt) CurTrackIndex = AUTOPLAYTRACKS[RandomInt,number] timer("track_update",15)}
        }
    }

    #Regular sequential autoplay mode
    else
    {
        #Stop autoplay when we've played everything
        if((AutoplayPlayed:count()+1)>AUTOPLAYTRACKCOUNT) {Autoplay = 0 AutoplayPlayed = array() stoptimer("autoplay_update") ownerNotify("Reached end of track list, stopping.")}

        else
        {
            #Play the next track
            AutoplayPlayed:pushNumber(AUTOPLAYTRACKS[AutoplayPlayed:count()+1,number])
            CurTrackIndex = AutoplayPlayed[AutoplayPlayed:count(),number]
            timer("track_update",15)
        }
    }
}



#ALPHANUM DISPLAY
elseif(clk("display_update"))
{
    if(ALPHANUM[2])
    {
        local TrackName = CurTrackTable["track_name",string]

        #Dont bother scrolling if it fits in the display
        if(TrackName:length()<=ALPHANUM[2])
        {
            stoptimer("display_update")

            ALPHANUM:updateAlphanumDisplay(TrackName:explode(""))
        }

        #Scrolling
        else
        {
            timer("display_update",DISPLAY_FREQUENCY)

            #Added space to the end for better scrolling
            TrackName = TrackName+" "
            local TrackNameExp = TrackName:explode("")
            local TrackNameLength = TrackName:length()

            ScrollIndex++

            if(ScrollIndex > TrackNameLength) {ScrollIndex=1}

            Scroll:pushString(TrackNameExp[ScrollIndex,string])

            if(Scroll:count()>TrackNameLength) {Scroll:remove(1)}

            ALPHANUM:updateAlphanumDisplay(Scroll)
        }
    }
}
