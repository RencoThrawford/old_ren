    if(changed(Power_On)|changed(Weapons_Free))
    {
        Active = (Power_On & Weapons_Free)
    
        setName("renFCS "+VERSION+"\n"+
        STATUSES[Active,string]+": "+(Active ? ACTIVE_FREQUENCY : IDLE_FREQUENCY)+"ms")
    
        if(Active) {timer("misc_update",100)}
    }

    #Update precached stuff on gun/ammunition change
    if(changed(Current_Gun)|changed(Current_Gun:acfMuzzleVel()))
    {
        CurMuzzleVel = Current_Gun:acfMuzzleVel()
        GLATGMOverride = (Current_Gun:acfAmmoType()=="GLATGM")
    
        #Move and parent the elevation reference hologram
        holoParent(ELEVREFI,Current_Gun)
        holoPos(ELEVREFI,Current_Gun:massCenter())
    
        if(CurMuzzleVel == 0)
        {
            Current_Table = MISSILETABLES[MISSILEKEY,array]
        }
    
        elseif(!GLATGMOverride)
        {
            Current_Table = renFCSLTableCreate(Current_Gun,TABLEREFI)
        }
    }
}



#Main ballistics loop
elseif(clk("ballistics_update"))
{
    timer("ballistics_update",Active ? ACTIVE_FREQUENCY : IDLE_FREQUENCY)

    #GLATGM Mode
    if(GLATGMOverride)
    {
        Solution = vec(0)
    }

    #Parabolic mode
    elseif(Parabolic_Mode)
    {
        Solution = vec(0)
    }

    #Regular range table lookup
    else
    {
        holoAng(ELEVREFI,ang(0,Current_Gun:angles():yaw(),0))

        GunRange = ((Current_Gun:massCenter() - Target_Pos):setZ(0):length())/RenFCSL_CONVERT
        local AdjustedRange = GunRange-abs(ELEVREFE:elevation(Target_Pos)/RenFCSL_CONVERT)

        Lookup = renFCSLTableInterpolate(Current_Table,AdjustedRange)

        Flight_Time = Lookup[2]

        local GunDrift = (Current_Vel * Flight_Time)
        local LeadDrift = (Target_Vel * Flight_Time)
        local Opposite = renFCSLFindOpposite(AdjustedRange,Lookup[1])

        Solution = vec(0,0,Opposite) - (GunDrift-LeadDrift)
    }

    msgBusOut()
}



#Main display loop
elseif(clk("misc_update"))
{
    if(Active) {timer("misc_update",MISC_FREQUENCY)}

    renFCSLUpdateDisplay(Range,Active)
}



#Update the MSGBUS with blank data upon E2 destruction (Hopefully)
elseif(last())
{
    Solution = vec(0)
    Flight_Time = 0

    msgBusOut()
}
