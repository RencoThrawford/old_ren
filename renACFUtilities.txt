    #Used for pulling out the numbers from a string

    #/////FUNCTIONS\\\\\#
    #[Prints a message in the owner's chat
    Inputs:
        Message [STRING] - The message that will be sent
    ]#
    function void printOwner(Message:string)
    {
        print("[renACFUtilities] "+Message)
    }

    #[Notifies the owner via a hint
    Inputs:
        Message [STRING] - The message that will be hinted
    ]#
    function void notifyOwner(Message:string)
    {
        hint("[renACFUtilities] "+Message,5.5)
    }

    #[Prints a message into the owner's console
    Inputs:
        Message [STRING] - The message that will be printed into console
    ]#
    function void bugOwner(Message:string)
    {
        print(_HUD_PRINTCONSOLE,"[renACFUtilities] "+Message)
    }

    #[Plays a computer processing sound thing
    ]#
    function void soundWorking()
    {
        soundPlay("working",15,"ambient/computer_working2.wav")
        soundVolume("working",0.75)
        soundVolume("working",0,15)
    }

    #[Plays an error sound
    ]#
    function void soundError()
    {
        soundPlay("error",0.5,"buttons/button11.wav")
        soundVolume("error",0.75)
    }

    #[Plays an confirmation sound
    ]#
    function void soundSelection()
    {
        soundPlay("selection",0.2,"buttons/blip1.wav")
        soundVolume("selection",0.5)
    }

    #[Creates a hologram on the given entity, scaled and coloured
    Inputs:
        [ENTITY] - The entity the hologram will be created for
    ]#
    function void entity:createHoloSelection()
    {
        holoCreate(0,This:pos(),vec(1.15),This:angles(),vec4(255,165,0,100),This:model())
        holoParent(0,This)
        holoMaterial(0,"models/ihvtest/eyeball_l")
    }

    #[Formats seconds into a vector (SS,MM,HH)
    Inputs:
        [NUMBER] - Time in seconds
    Return:
        [VECTOR] - Time in seconds, minutes, hours vec(S,M,H)
    ]#
    function vector number:formatTime()
    {
        local Seconds = floor(This%60)
        local Minutes = floor(This/60)
        local Hours = floor(Minutes/60)
        Minutes = floor(Minutes%60)
        
        return vec(Seconds,Minutes,Hours)
    }

    #[For prettyness and ACF 3 forward compatability
    Inputs:
        [ENTITY] - The entity to check
    Return:
        [NUMBER] - Maximum torque with fuel (In Nm)
    ]#
    function number entity:maxTorqueWithFuel()
    {
        #ifdef entity:acfMaxTorqueWithFuel()
            return This:acfMaxTorqueWithFuel()
        #else
            return -1
        #endif
    }
    
    #[Computes the probability of overmatch given a prop and caliber
    Inputs:
        [ENTITY] - The prop that will be checked
        Caliber [NUMBER] - The caliber of weapon that will be checked for
    ]#
    function number entity:overmatchCheck(Caliber:number)
    {
        return clamp((Caliber/This:acfPropArmorMax() - 1.3)/(7-1.3),0,1)
    }
    
    #[The beef of this thing, does everything
    Inputs:
        [ENTITY] - The entity to be chekced
        Arg [STRING] - An argument.
    ]#
    function void entity:checkEnt(Arg:string)
    {
        local Type = This:type()
        switch(Type)
        {
            case "acf_engine",
                local Links = This:acfLinks()
                bugOwner("Found "+Links:count()+" fuel tanks/batteries.")
                if(Links:count())
                {
                    local FuelTotal = 0
                    foreach(K,L:entity = Links) {if(L:acfIsFuel()) {FuelTotal += L:acfFuel()}}

                    local FuelUse = This:acfPeakFuelUse()
                    local Duration = (FuelTotal/FuelUse)*60
                    local HMSDuration = Duration:formatTime()
                    local Units = "L"
                    if(This:acfType() == "Electric") {Units = "kW"}

                    printOwner("Fuel time at max usage - "+HMSDuration[3]+":"+HMSDuration[2]+":"+HMSDuration[1]+" @ "+FuelUse+Units+"/m, w/"+FuelTotal+Units+".")
                    soundWorking()
                }
                else{notifyOwner("No fuel tanks/batteries detected!") soundError()}
            break

            case "acf_gearbox",
                local Links = This:acfLinks()
                local Engines = array()
                local Wheels = array()
                foreach(K,E:entity = Links) {if(E:acfIsEngine()) {Engines:pushEntity(E)} else{Wheels:pushEntity(E)}}
                bugOwner("Found "+Engines:count()+" engine(s) and "+Wheels:count()+" wheel(s).")

                if(Engines:count())
                {
                    local EngineTorque = 0
                    local EngineTorqueWFuel = 0
                    foreach(I,E:entity = Engines) {EngineTorque += E:acfMaxTorque() EngineTorqueWFuel += E:maxTorqueWithFuel()}
                    local GearboxMaxTorque = This:acfTorqueRating()
                    local TorqueAssesment = OVERLOADS[EngineTorque>GearboxMaxTorque,string]
                    local TorqueFuelAssesment = OVERLOADS[EngineTorqueWFuel>GearboxMaxTorque,string]
                    
                    printOwner("No fuel: "+TorqueAssesment+" @ "+EngineTorque+" Nm, Fuel: "+TorqueFuelAssesment+" @ "+EngineTorqueWFuel+" Nm; "+GearboxMaxTorque+" Nm.")
                    soundWorking()
                }
                else{notifyOwner("No engines detected!") soundError()}
            break

            case "prop_physics",
                local Weapon = Arg
                if(!Arg) {Weapon = "120c"} #Default to 120mm cannon
                if(PENS:exists(Weapon))
                {
                    local CaliberS = ""
                    foreach(K,N:string = Weapon:explode("")) {if(NUMBERS:exists(N)) {CaliberS += N}} #Pulls out the numbers and periods to make a number
                    local Caliber = CaliberS:toNumber()
                    local MaxPen = PENS[Weapon,number]
                    local OvermatchProb = This:overmatchCheck(Caliber)
                    
                    printOwner("Overmatch probability "+round(OvermatchProb*100,3)+"% w/"+Weapon+" @ "+MaxPen+"mm pen; "+This:acfPropArmorMax()+"mm armor.")
                    soundWorking()
                }
                else {notifyOwner(Arg+" is not a valid weapon, please check the E2 for a list of valid weapons.") soundError()}
            break

            default,
                notifyOwner(Type+" is not an entity supported by this E2.")
                soundError()
            break
        }
    }



    #/////FINALIZE\\\\\#
    #//MISC\\#
    runOnChat(1)


    #//VERSION\\#
    VERSION = "V1.0.0"
    setName("renACFUtilities "+VERSION)


    #//CHECK-ON-SPAWN\\#
    Ent = entity():isWeldedTo()
    if(Ent:isValid()) {Ent:checkEnt("")} else{notifyOwner("Use 'help' for a list of commands and 'manual <COMMAND>' to learn how to use them.")}
}



#//////////CHAT COMMANDS\\\\\\\\\\#
elseif(chatClk(owner()))
{
    local Said = owner():lastSaid():lower()
    local Split = Said:explode(" ")
    local Command = Split[1,string]
    local Arg = Split[2,string]

    switch(Command)
    {
        #manual - Prints out useful information about a command
        case "manual",
            if(MANUAL:exists(Arg)) {printOwner(Arg+": "+MANUAL[Arg,string])}
            else{notifyOwner("No manual entry found for "+Arg+", please use 'help' for a list of commands.")}
            hideChat(1)
        break

        #help - Provides a list of available commands
        case "help",
            printOwner("Commands: manual, help, version, reset, check, set, remove.")
            hideChat(1)
        break

        #version - Prints out the version of the E2
        case "version",
            printOwner(VERSION)
            hideChat(1)
        break

        #reset - Resets the E2
        case "reset",
            notifyOwner("Resetting.")
            hideChat(1)
            reset()
        break

        #check - Performs a check of the selected entity
        case "check",
            Ent:checkEnt(Arg)
            hideChat(1)
        break

        #set - Changes the currently selected entity
        case "set",
            if(Arg)
            {
                Ent = entity(Arg:toNumber())
            }
            else
            {
                Ent = owner():aimEntity()
            }
            
            notifyOwner("Entity now set to "+Ent:type()+" ["+Ent:id()+"].")
            
            hideChat(1)
        break

        #remove - Removes the e2
        case "remove",
            notifyOwner("Removing E2.")
            hideChat(1)
            selfDestruct()
        break
    }
}

    "Selected: "+Ent:type()+" ["+Ent:id()+"]")
}
