    #Grabs everything this E2 needs from the MSGBUS
    function void msgBusIn()
    {
        if(->MSGBUS)
        {
            Self = MSGBUS["SELF",array]
            FLIR = MSGBUS["FLIR",number]
            Operator = MSGBUS["GUNNER",entity]
        }
    }


    #//MISC\\#
    #[Hints to the operator, or the owner, the message
    Inputs:
        Message [STRING] - Notifies either the Operator or the Owner depending on presence
    ]#
    function void operatorNotify(Message:string)
    {
        if(Operator:isPlayer()) {Operator:vehicle():hintDriver("[renCamo] "+Message,5.5)}
        if(Operator!=owner()) {hint("[renCamo] "+Message,5.5)}
    }

    #[Prints to the operator, or the owner, the message
    Inputs:
        Message [STRING] - Prints to the chat of either the operator or the Owner
    ]#
    function void operatorTell(Message:string)
    {
        if(Operator:isPlayer()) {Operator:vehicle():printDriver("[renCamo] "+Message)}
        if(Operator!=owner()) {print("[renCamo] "+Message)}
    }
    
    function number entity:checkSphericalWheel()
    {
        local Radius = This:radius()
        if(Radius < 10) {return 0}
        local Box = This:boxSize()
        if(Box[1] == Box[2])
        {
            if(Box[2] == Box[3])
            {
                if(Box[3] == Box[1])
                {
                    return 1
                }
                else
                {
                    return 0
                }
            }
            else
            {
                return 0
            }
        }
        else
        {
            return 0
        }
    }
