POSITION                          [VECTOR3] - Current vehicle position

TARGET_SPEED                      [VECTOR3] - Speed of target for ballistics computer if not manually leading
GUNNER                            [ENTITY]  - Gunner controlling armament

FLIR                              [NUMBER]  - If FLIR on/off

NODE_LISTING                      [ARRAY]   - BUS Nodes

    function void notifyGunner(Message:string)
    {
        Gunner:vehicle():hintDriver("[renOptics] "+Message,5.5)

    #[Formats a string with Gunner: <GUNNERNAME>, Driver: <DRIVERNAME>. depending on who's here
    Inputs:
        Gunner [ENTITY] - The Gunner
        Driver [ENTITY] - The Driver
    Return:
        [STRING] Returns a string with either the gunner or driver
    ]#
    function string formatRIFFL(Gunner:entity,Driver:entity)
    {
        local Return = array()
        if(Gunner:isPlayer()) {Return:pushString("Gunner: "+Gunner:name())}
        if(Driver:isPlayer()&(Driver!=Gunner)) {Return:pushString("Driver: "+Driver:name())}
        return vonEncode(Return)
    }

    #//MSGBUS\\#
    #Since this E2 acts as the "hub" we need to create each and every one of the thingies for thingies to work
    MSGBUS = table()

#//////////ZOOM\\\\\\\\\\#

elseif(inputClk())
{
    #If we scroll and the camera is active, zoom in/out
    if(changed(NextWeapon|PrevWeapon)&(!Cam_Off))
    {
        #Increasing the zoom factor
        local ZoomInc = (NextWeapon-PrevWeapon)*20
        Zoom = clamp(Zoom+ZoomInc,10,300)

        #Clamping the FOV and CamDist
        local FOV = clamp(Zoom,10,90)
        local CamDist = clamp(Zoom,90,300)-90 #-90 So it starts at 0

        #Update those thangs
        Cam["Distance",number] = max(CamDist,0)
        Cam["FOV",normal] = FOV
        if(USINGGFS) {GFSCam["Distance",number] = max(CamDist,0) GFSCam["FOV",number] = FOV}

        Gunner:renOLUpdateVisibility(!Cam_Off,CamDist>0)
    }
}



#//////////CHAT COMMANDS\\\\\\\\\\#
elseif(chatClk(Gunner))
{
    local Said = Gunner:lastSaid():lower()
    local Split = Said:explode(" ")
    local Command = Split:string(1)
    local Arg1 = Split:string(2)

    switch(Command)
    {
        case "fs",
            foreach(K,E:entity = Self)
            {
                #ifdef entity:propFreeze(number)
                    E:propFreeze(Arg1:toNumber())
                #endif
            }

            hideChat(1)
        break

        case "nodes",
            print(MSGBUS["NODE_NAMES",array])

            hideChat(1)
        break

        default,
            scr_chat(GunnerSeat, rangerOffset(9999999999,CAMREFE:pos()+Cam["CamDir",vector],Cam["CamDir",vector]):pos())
        break
    }
}



#//////////KEYS\\\\\\\\\\#
#All of the keyey goodness
elseif(keyClk(Gunner))
{
    #Don't do key things when the user is typing
    if((!Gunner:isTyping())&Gunner:vehicle())
    {
        #Get the things
        local Key = keyClkPressed()
        local Pressed = Gunner:keyPressed(Key)

        #/////Keys\\\\\#
        switch(Key)
        {
            #//VIEW SWITCH\\#
            case RenKL_KEYS["VIEW_SWITCH",string],
                if(Pressed)
                {
                    Cam_Off = !Cam_Off

                    Cam["Activated",number] = !Cam_Off

                    Gunner:renOLUpdateVisibility(!Cam_Off,(clamp(Zoom,90,300)-90)>0)
                }
            break

            #//FLIR TOGGLE\\#
            case RenKL_KEYS["SWITCH_AIMER",string],
                if(Pressed)
                {
                    if(Aimer+1 > AIMERS:count()) {Aimer = 1} else{Aimer++}

                    switch(Aimer)
                    {
                        case 1,
                            RenOL_AIMERI:renOLCreateDayAimer(Gunner)
                            FLIR = 0

                            soundPlay("aimer switch",1.1,"buttons/button1.wav")
                        break

                        case 2,
                            RenOL_AIMERI:renOLCreateNightAimer(Gunner)
                            FLIR = 1

                            soundPlay("aimer switch",1.1,"buttons/button1.wav")
                            soundVolume("aimer switch",0.5)
                        break

                        case 3,
                            RenOL_AIMERI:renOLCreateBackupAimer(Gunner,Range_Table)
                            FLIR = 0

                            soundPlay("aimer switch",1.1,"buttons/button1.wav")
                        break
                    }

                    notifyGunner("Aimer switched to "+AIMERS[Aimer,string]+".")

                    Cam["FLIR",number] = FLIR
                    GFSCam["FLIR",number] = FLIR
                }
            break

            #//PARABOLIC TOGGLE\\#
            case RenKL_KEYS["TOGGLE_PARA",string],
                if(Pressed)
                {
                    Ballistics_Parabolic_On = !Ballistics_Parabolic_On

                    notifyGunner("Toggling parabolic "+ACTIVATIONS[Ballistics_Parabolic_On,string]+".")

                    soundPlay("parabolic toggle",0.3,"buttons/blip1.wav")
                    soundVolume("parabolic toggle",0.5)
                }
            break

            #//SOURCE CYCLE\\#
            case RenKL_KEYS["SOURCE_CYCLE",string],
                if(Pressed)
                {
                    if(CurSource+1 > SOURCES:count()) {CurSource = 1} else{CurSource++}

                    notifyGunner("Switching to "+SOURCES[CurSource,string]+".")

                    soundPlay("source switch",0.4,"buttongs/button9.wav")
                    soundVolume("source",0.5)
                }
            break

            #//COUNTERMEASURES DEPLOYATION\\#
            case RenKL_KEYS["DEPLOY_COUNTER",string],
                Countermeasures = Pressed
            break

            #//IFF INTERROGATION\\#
            case RenKL_KEYS["INTERROGATE",string],
                if(Pressed)
                {
                    notifyGunner("Interrogating target.")
                    
                    RIFFLEData = formatRIFFL(Gunner,Driver)

                    local RIFFLE = renIFFLStart(RIFFLEData,Cam["HitPos",vector],Curtime)

                    scr_transmit(6,RIFFLE[1,string],RIFFLE[2,vector],noentity())
                }
            break

            #//LASING/GFS\\#
            case RenKL_KEYS["LASE",string],
                if(USINGGFS)
                {
                    GFSActive = Pressed
                    
                    Cam["Activated",number] = !GFSActive
                    GFSCam["Activated",number] = GFSActive
                    
                    if(GFSActive) {} else{}
                }
            break

            #//SCRL INTEGRATION\\#
            default,
                if(Pressed)
                {
                    local Num = Key:right(1):toNumber()
                    if(Key:left(3) == "pad" & (Num | Key:right(1)=="0"))
                    {
                        scr_keyTransmit(Gunner, Cam["HitPos",vector])
                    }
                }
            break
        }
    }
}



#//////////BUS UPDATE\\\\\\\\\\#
#Sends and receives things from the bus, and also gets the aimposition and range
elseif(clk("bus_update"))
{
    #Start the timer again, and get everything from the bus
    timer("bus_update",BUS_UPDATE_FREQUENCY)
    msgBusIn()

    #Updating the Aimpos
    switch(CurSource)
    {
        #//REGULAR AIMPOS\\#
        case 1,
            if(!GFSActive)
            {
                Aimpos = Cam["HitPos",vector]
                Target_Speed = vec(0)
                Target_Track = 0.25
            }
        break

        #//SCRL ARTILLERY COMMAND\\#
        case 2,
            Aimpos = ArtyRequest
            Target_Speed = vec(0)
            Target_track = 1
        break

        #//GTMP MAP POINT\\#
        case 3,
            Aimpos = Map_Point
            Target_Speed = vec(0)
            Target_Track = 1
        break

        #//SPOTTER DRONE TARGET\\#
        case 4,
            Aimpos = Drone_Target
            Target_Speed = vec(0)
            Target_Track = 0.25
        break

        #//CIWS TARGET\\#
        case 5,
            Aimpos = Counter_Target
            Target_Speed = Counter_Speed
            Target_Track = 1
        break

        #//TESET TARGET\\#
        case 6,
            Aimpos = Test_Target
            Target_Speed = $Test_Target
            Target_Track = 1
        break
    }

    #Get the range from the camera reference to the aimpos, converts to meters
    Range = ((CAMREFE:pos() - Aimpos):setZ(0):length())/39.37

    #Update all the positionals, curtime
    Position = Base:pos()
    Angle = Base:angles()
    Velocity = Base:vel()
    Curtime = curtime()

    #Automatically turn on parabolic mode if the main gun is parabolic
    if(changed(Main_Gun)) {Ballistics_Parabolic_On = Main_Gun_Parabolic}

    #Output everything
    msgBusOut()
}

#//////////GUNNER QUERY\\\\\\\\\\#
#Looks for a gunner.
elseif(clk("gunner_query"))
{
    #Start the timer over again and check to see if anyone took the bait
    timer("gunner_query",(Gunner) ? (GUNNER_QUERY_FREQUENCY) : (GUNNER_QUERY_FREQUENCY >> 1))

    Gunner = GunnerSeat:driver()

    if(changed(Gunner))
    {
        #Update the visbility for the gunner
        Gunner:renOLUpdateVisibility(!Cam_Off,(clamp(Zoom,90,300)-90)>0)

        #Update RIFFLEData
        RIFFLEData = formatRIFFL(Gunner,Driver)

        #We've got someone, turn on the important things, and return the camera to it's previous state
        if(Gunner:isPlayer())
        {
            timer("cam_ref",CAMREF_FREQUENCY)
            timer("aimer_update",AIMER_FREQUENCY)

            Cam["Activated",number] = !Cam_Off
        }

        #They left, stop those times and shut the camera off juuuust in case
        else
        {
            stoptimer("cam_ref")
            stoptimer("aimer_update")

            Cam["Activated",number] = 0
        }
    }
}
